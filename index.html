function obtenerInformacion(clase) { const texto = clase .toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .trim(); // NO RECICLABLES if ( texto.includes("no reciclable") || texto.includes("no reciclables") || texto.includes("noreciclable") || texto.includes("noreciclables") ) { return { nombre: clase, categoria: "No reciclables", bote: "🚫 Bote de no reciclables" }; } // ESPECIALES if ( texto.includes("especial") || texto.includes("especiales") ) { return { nombre: clase, categoria: "Especiales", bote: "🔋 Punto de recolección especial" }; } // PLÁSTICO if (texto.includes("plast")) { return { nombre: clase, categoria: "Plástico", bote: "🟡 Bote amarillo" }; } // ORGÁNICOS if (texto.includes("organ")) { return { nombre: clase, categoria: "Orgánicos", bote: "🟢 Bote verde" }; } // PAPEL Y CARTÓN if ( texto.includes("papel") || texto.includes("carton") ) { return { nombre: clase, categoria: "Papel y cartón", bote: "🔵 Bote azul" }; } // VIDRIO if (texto.includes("vidrio")) { return { nombre: clase, categoria: "Vidrio", bote: "🟢 Bote verde" }; } // METALES if (texto.includes("metal")) { return { nombre: clase, categoria: "Metales", bote: "🟡 Bote amarillo" }; } // ELECTRÓNICOS if ( texto.includes("elect") || texto.includes("electron") ) { return { nombre: clase, categoria: "Electrónicos", bote: "🔌 Punto de recolección electrónica" }; } return { nombre: clase, categoria: clase, bote: "🗑️ Verificar el recipiente correspondiente" }; }
